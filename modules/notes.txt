1. Start PostgreSQL with Docker

	docker compose -p postgresql up -d

2. Restore NuGet Packages

	dotnet restore
	
3. Run the application

	dotnet run --project MovieApi.Api

4. pgAdmin is available at http://localhost:5050

		Password:						dbadminsecret

	Servers:

		Register > Server...

			General
				Name: PostgresLocal

			Connection
				Host name/address: postgres
				Port: 5432
				Maintenance database: dotnetHero
				Username: dbuser
				Password: dbusersecret

		PostgresLocal > Databases > dotnetHero > Query Tool

			-- Offset pagination — gets slower as offset grows
			SELECT *
			FROM app."Movies"
			WHERE "Title" ILIKE '%action%' OR "Genre" ILIKE '%action%'
			ORDER BY "Rating" DESC, "Title" ASC, "Id" ASC
			LIMIT 5 OFFSET 0;  -- page 1, pageSize 5
	
			-- Keyset pagination — constant speed regardless of position
			SELECT *
			FROM app."Movies"
			WHERE "Created" > '2026-01-01T00:00:00Z'
			ORDER BY "Created" ASC, "Id" ASC
			LIMIT 10;
